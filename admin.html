// Banner muncul SEKALI per sesi function showWelcomeBanner(username, roleDisplay, role) { const banner = document.getElementById('welcomeBanner'); if (!banner) return; // Cek: udah pernah muncul di sesi ini? if (sessionStorage.getItem('arcane-welcome-shown') === 'yes') { banner.remove(); return; } const usernameEl = document.getElementById('welcomeUsername'); const roleEl = document.getElementById('welcomeRole'); usernameEl.textContent = username; if (role === 'owner') { roleEl.style.display = 'none'; } else { roleEl.textContent = `(${roleDisplay})`; roleEl.style.display = 'inline-block'; } sessionStorage.setItem('arcane-welcome-shown', 'yes'); setTimeout(() => banner.classList.add('visible'), 100); setTimeout(() => { banner.classList.remove('visible'); setTimeout(() => banner.remove(), 600); }, 3000); }
